)
FetchContent_MakeAvailable(unity)

# One executable per test file; each is registered with CTest.
function(fc_core_test name)
    add_executable(${name} ${name}.c)
    target_link_libraries(${name} PRIVATE fc_core unity)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

# Same test source built against the 4 kHz variant.
function(fc_core_test_4k name)
    add_executable(${name}_4k ${name}.c)
    target_link_libraries(${name}_4k PRIVATE fc_core_4k unity)
    target_compile_options(${name}_4k PRIVATE -Wall -Wextra -Werror)
    add_test(NAME ${name}_4k COMMAND ${name}_4k)
endfunction()
